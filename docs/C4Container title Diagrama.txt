C4Container
title Diagrama C4 - Nivel Contenedores - Mini Jira

Person(usuario, "Usuario normal", "Crea, consulta, comenta, actualiza y archiva tickets según sus permisos.")
Person(admin, "Administrador", "Gestiona todos los tickets, usuarios, dashboard global, auditoría y restauración de tickets archivados.")

System_Boundary(minijira, "Mini Jira") {

  Container(frontend, "Frontend Web", "React + TypeScript + Material UI", "Interfaz web responsive para tablero, lista de tickets, detalle, comentarios, filtros, dashboard y gestión básica de usuarios.")

  Container(api, "Backend API", "Node.js + TypeScript + NestJS/Express", "Expone endpoints REST, valida autenticación, permisos, reglas de negocio, concurrencia optimista y orquesta los módulos del sistema.")

  Container(auth, "Módulo de Autenticación y Autorización", "Backend Module", "Gestiona login, logout, sesión/JWT, roles admin/user y autorización por permisos en backend.")

  Container(ticketService, "Módulo de Tickets", "Backend Module", "Gestiona creación, edición, asignación, cambio de estado, filtros, archivado lógico, restauración y control de versión.")

  Container(commentService, "Módulo de Comentarios", "Backend Module", "Registra comentarios, autores, fechas, historial y detecta menciones con formato @usuario.")

  Container(notificationService, "Módulo de Notificaciones", "Backend Module", "Genera eventos de email por asignación, mención y cambio a estado blocked, evitando duplicados por reintentos.")

  Container(dashboardService, "Módulo de Dashboard", "Backend Module", "Calcula métricas de tickets por estado, cerrados por mes, activos por responsable, bloqueados y tiempo de cierre respetando permisos.")

  Container(auditService, "Módulo de Auditoría", "Backend Module", "Registra eventos críticos: creación, cambios de estado, responsable, prioridad, título, descripción, archivado y restauración.")

  ContainerDb(database, "Base de Datos", "PostgreSQL", "Almacena usuarios, roles, tickets, comentarios, logs de auditoría y eventos de notificación.")

  ContainerDb(mailhog, "Servidor SMTP / Proveedor Email", "SMTP / Nodemailer / Mailhog", "Entrega correos transaccionales de asignaciones, menciones y tickets bloqueados.")
}

Rel(usuario, frontend, "Usa", "HTTPS")
Rel(admin, frontend, "Usa", "HTTPS")

Rel(frontend, api, "Consume API REST", "JSON/HTTPS")

Rel(api, auth, "Valida identidad, sesión y permisos")
Rel(api, ticketService, "Orquesta operaciones de tickets")
Rel(api, commentService, "Orquesta comentarios")
Rel(api, dashboardService, "Consulta métricas")
Rel(api, auditService, "Registra cambios auditables")
Rel(api, notificationService, "Solicita generación de eventos de email")

Rel(auth, database, "Lee usuarios, roles y credenciales", "SQL")
Rel(ticketService, database, "Lee y escribe tickets con versionado optimista", "SQL")
Rel(commentService, database, "Lee y escribe comentarios", "SQL")
Rel(dashboardService, database, "Consulta agregados y métricas", "SQL")
Rel(auditService, database, "Escribe logs de auditoría", "SQL")
Rel(notificationService, database, "Registra eventos enviados/fallidos y evita duplicados", "SQL")

Rel(commentService, notificationService, "Genera evento user_mentioned")
Rel(ticketService, notificationService, "Genera eventos ticket_assigned y ticket_blocked")
Rel(ticketService, auditService, "Registra creación, edición, estado, responsable, archivado y restauración")
Rel(commentService, auditService, "Puede registrar cambios relevantes de comentarios")

Rel(notificationService, mailhog, "Envía emails transaccionales", "SMTP")
